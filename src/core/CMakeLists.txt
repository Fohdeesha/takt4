# takt4_assets — the weights and the state space, compiled into the program.
#
# What makes takt4.exe one file somebody can copy to a stick and run. HANDOFF §8 Phase 7
# had an installer put these beside the executable; inside it is the same intent taken one
# step further, and it removes the failure the copy on D:\ actually hit — an exe that went
# on reading its weights out of the source tree it happened to be built in.
#
# Kept out of takt4_core deliberately. The console and the test binary link core, want to be
# pointed at *any* weight set rather than the one that is built in, and have no reason to
# carry three megabytes of literal in every link. Only the application links this.
#
# Python is required only when the UI is built; a core-only configuration never runs it.
if(TAKT4_BUILD_UI)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)

  set(takt4_embed_script "${PROJECT_SOURCE_DIR}/tools/embed_asset.py")
  set(takt4_assets_dir "${PROJECT_BINARY_DIR}/generated/assets")

  # (function name, asset file) pairs. The script rewrites its output only when the bytes
  # change, so an incremental build does not recompile a multi-megabyte translation unit
  # every time something else moves.
  # `electronic` is the built-in set since 0.9.1: BeatNet+'s `generic` fine-tuned on the
  # operator's own library (TRACKING-PROPOSAL.md §7.13). The three published sets stay in
  # assets/weights/ for the console and the tests.
  #
  # The licence and the third-party notices go in too: takt4 is one file, and the notices of
  # what is built into it have to travel with it (GPLv3 §5(d), and the BSD, MIT, Apache, MPL
  # and Unicode licences of the rest). The About box opens them.
  #
  # (name|file|sidecar): a blob with a sidecar must have the sha256 the sidecar records.
  foreach(entry "weights|${PROJECT_SOURCE_DIR}/assets/weights/electronic.bin|${PROJECT_SOURCE_DIR}/assets/weights/electronic.json"
                "stateSpace|${PROJECT_SOURCE_DIR}/assets/statespace/default.bin|${PROJECT_SOURCE_DIR}/assets/statespace/default.json"
                "licence|${PROJECT_SOURCE_DIR}/LICENSE|"
                "notices|${PROJECT_SOURCE_DIR}/THIRD-PARTY-NOTICES.txt|")
    string(REPLACE "|" ";" parts "${entry}")
    list(GET parts 0 fn)
    list(GET parts 1 blob)
    list(GET parts 2 manifest)
    set(generated "${takt4_assets_dir}/${fn}_blob.cpp")
    set(manifest_args "")
    set(manifest_deps "")
    if(manifest)
      set(manifest_args --manifest "${manifest}")
      set(manifest_deps "${manifest}")
    endif()
    add_custom_command(
      OUTPUT "${generated}"
      COMMAND "${Python3_EXECUTABLE}" "${takt4_embed_script}" "${blob}" "${generated}"
              --name "${fn}" ${manifest_args}
      DEPENDS "${blob}" "${takt4_embed_script}" ${manifest_deps}
      COMMENT "Embedding ${blob}"
      VERBATIM
    )
    list(APPEND takt4_asset_sources "${generated}")
  endforeach()

  add_library(takt4_assets STATIC ${takt4_asset_sources} assets/embedded.hpp)
  add_library(takt4::assets ALIAS takt4_assets)
  target_link_libraries(takt4_assets PRIVATE takt4::core takt4::warnings)
endif()

# The real-time allocation guard (HANDOFF §7.6): rt/alloc_guard_new.cpp replaces the
# global operator new/delete. A replacement has to be compiled into the executable
# itself — one sitting in a static library is not guaranteed to be pulled in — so it
# travels as an INTERFACE source, and takt4_enable_rt_guard() adds it to a target
# together with the TAKT4_RT_ALLOC_GUARD definition that switches it on.
add_library(takt4_rt_guard INTERFACE)
add_library(takt4::rt_guard ALIAS takt4_rt_guard)
target_sources(takt4_rt_guard INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/rt/alloc_guard_new.cpp")
target_link_libraries(takt4_rt_guard INTERFACE takt4::core)

# takt4_enable_rt_guard(<executable target> ALWAYS|DEBUG_ONLY)
function(takt4_enable_rt_guard target when)
  target_link_libraries(${target} PRIVATE takt4::rt_guard)
  if(when STREQUAL "ALWAYS")
    target_compile_definitions(${target} PRIVATE TAKT4_RT_ALLOC_GUARD=1)
  elseif(when STREQUAL "DEBUG_ONLY")
    target_compile_definitions(${target} PRIVATE "TAKT4_RT_ALLOC_GUARD=$<IF:$<CONFIG:Debug>,1,0>")
  else()
    message(FATAL_ERROR "takt4_enable_rt_guard: expected ALWAYS or DEBUG_ONLY, got '${when}'")
  endif()
endfunction()
